    /* Capsules holding a vehicle. Cargo travels inside as the original items: in the container, where its full weight
       counts (no WeightReduction), or, packed while sandbox CargoWeight is off, in the Resources component (resource
       `cargo`), which the engine saves with the item like a container's contents but leaves out of the weight
       (InventoryContainer.getContentsWeight reads the container only; shared/MinidoracatPackThrow/Core.lua M.cargoStore).
       MaxItemSize keeps players from putting anything in; the server adds cargo with AddItem, which skips the check.
       Container, not Tool: Minidoracat Economy rebuilds traded items with a new item ID (the record key), and its
       default and production whitelists leave Container out; server/Economy.lua registers each type with Economy
       (API rev 6) or blocks it (rev 5). */

    item LoadedCapsuleCartoon
    {
        DisplayCategory = Container,

        WorldStaticModel = MinidoracatPackThrow.PackThrowCartoonLoaded,
        StaticModel = MinidoracatPackThrow.PackThrowCartoonLoadedHand,
        Capacity = 1,
        MaxItemSize = 0.01,

        component Resources
        {
            cargo = Item@Input@1000000@StackAny,
